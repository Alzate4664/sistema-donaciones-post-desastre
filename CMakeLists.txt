cmake_minimum_required(VERSION 3.20)
project(Donaciones LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /utf-8 /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(donaciones src/modelo.cpp src/red.cpp src/sistema.cpp src/persistencia.cpp)
target_include_directories(donaciones PUBLIC include)
add_executable(pruebas tests/pruebas.cpp)
target_link_libraries(pruebas PRIVATE donaciones)
enable_testing()
add_test(NAME pruebas COMMAND pruebas)
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/gui_win32.cpp")
  add_executable(donaciones_gui WIN32 src/gui_win32.cpp)
  target_compile_definitions(donaciones_gui PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(donaciones_gui PRIVATE donaciones comctl32 comdlg32)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/bench/benchmark.cpp")
  add_executable(benchmark bench/benchmark.cpp)
  target_link_libraries(benchmark PRIVATE donaciones)
endif()
